set(TARGET_NAME caors)

add_library(${TARGET_NAME})
add_library(${TARGET_NAME}::${TARGET_NAME} ALIAS ${TARGET_NAME})

include(GNUInstallDirs)

set(CAORS_GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/include")
set(CAORS_GENERATED_VERSION_HEADER
    "${CAORS_GENERATED_INCLUDE_DIR}/caors/version.h")

file(MAKE_DIRECTORY "${CAORS_GENERATED_INCLUDE_DIR}/caors")
configure_file(
  "${CAORS_ROOT_DIR}/include/caors/version.h.in"
  "${CAORS_GENERATED_VERSION_HEADER}"
  @ONLY
)

file(GLOB_RECURSE LIB_SOURCE CONFIGURE_DEPENDS "*.cpp")
file(
  GLOB_RECURSE LIB_PUBLIC_HEADERS
  CONFIGURE_DEPENDS
  "${CAORS_ROOT_DIR}/include/*.h"
  "${CAORS_ROOT_DIR}/include/*.hpp"
)

target_sources(
  ${TARGET_NAME}
  PRIVATE ${LIB_SOURCE}
  PUBLIC
    FILE_SET HEADERS
    BASE_DIRS
      ${CAORS_ROOT_DIR}/include
      ${CAORS_GENERATED_INCLUDE_DIR}
    FILES
      ${LIB_PUBLIC_HEADERS}
      ${CAORS_GENERATED_VERSION_HEADER}
)

target_compile_features(${TARGET_NAME} PUBLIC cxx_std_20)
target_compile_warnings(${TARGET_NAME})

target_compile_options(
  ${TARGET_NAME}
  PUBLIC
    "$<$<AND:$<COMPILE_LANGUAGE:C>,$<C_COMPILER_ID:MSVC>>:/utf-8>"
    "$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CXX_COMPILER_ID:MSVC>>:/utf-8>"
)

target_include_directories(
  ${TARGET_NAME}
  PUBLIC
    $<BUILD_INTERFACE:${CAORS_GENERATED_INCLUDE_DIR}>
    $<BUILD_INTERFACE:${CAORS_ROOT_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_compile_definitions(
  ${TARGET_NAME}
  PRIVATE
    CAORS_BUILDING_LIBRARY
  PUBLIC
    $<$<NOT:$<BOOL:${BUILD_SHARED_LIBS}>>:CAORS_STATIC_DEFINE>
    CAORS_USE_DEBUG_OUTPUT=$<BOOL:${USE_DEBUG_OUTPUT}>
    CAORS_USE_LOG=$<BOOL:${USE_LOG}>
    CAORS_USE_LOG_SEVERITY=${USE_LOG_SEVERITY}
)

set_target_properties(${TARGET_NAME} PROPERTIES FOLDER "Core")

install(
  TARGETS ${TARGET_NAME}
  EXPORT caorsTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  FILE_SET HEADERS DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

install(
  EXPORT caorsTargets
  NAMESPACE ${TARGET_NAME}::
  DESTINATION ${CAORS_INSTALL_CMAKEDIR}
)
