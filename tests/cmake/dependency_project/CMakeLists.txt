cmake_minimum_required(VERSION 3.23)
project(caors_dependency_contract LANGUAGES C CXX)
include("${CAORS_SOURCE_DIR}/cmake/caors_dependency_catalog.cmake")
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(CAORS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CAORS_BUILD_EXAMPLE_TESTING OFF CACHE BOOL "" FORCE)
if(CAORS_DEPENDENCY_CASE MATCHES "^ctrack_")
  set(CAORS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(CAORS_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
  add_library(host_ctrack INTERFACE)
  add_library(ctrack::ctrack ALIAS host_ctrack)
  if(CAORS_DEPENDENCY_CASE STREQUAL "ctrack_complete")
    set(ctrack_VERSION "${CAORS_CTRACK_VERSION}")
  elseif(CAORS_DEPENDENCY_CASE STREQUAL "ctrack_incompatible")
    set(ctrack_VERSION 1.0.2)
  endif()
else()
  set(CAORS_BUILD_TESTING ON CACHE BOOL "" FORCE)
  set(CAORS_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  # 只验证目标复用契约，不依赖 CAORS 私有的第三方源码布局。
  add_library(host_catch INTERFACE)
  add_library(Catch2::Catch2 ALIAS host_catch)
  if(NOT CAORS_DEPENDENCY_CASE STREQUAL "base_only")
    add_library(host_catch_main INTERFACE)
    add_library(Catch2::Catch2WithMain ALIAS host_catch_main)
    target_link_libraries(host_catch_main INTERFACE Catch2::Catch2)
  endif()
  if(CAORS_DEPENDENCY_CASE STREQUAL "complete")
    set(Catch2_VERSION "${CAORS_CATCH2_VERSION}")
  elseif(CAORS_DEPENDENCY_CASE STREQUAL "incompatible")
    set(Catch2_VERSION 2.13.0)
  endif()
endif()
set(CAORS_DEPENDENCY_OFFLINE ON CACHE BOOL "" FORCE)
set(CAORS_DEPENDENCY_SOURCE_CACHE_DIR "${CMAKE_CURRENT_BINARY_DIR}/unused-cache" CACHE PATH "" FORCE)
add_subdirectory("${CAORS_SOURCE_DIR}" caors)
if(EXISTS "${CAORS_DEPENDENCY_SOURCE_CACHE_DIR}")
  message(FATAL_ERROR "父目标复用不应准备源码缓存")
endif()
if(TARGET caors_catch OR TARGET caors_catch_main OR TARGET caors_ctrack)
  message(FATAL_ERROR "不应重复创建父项目已提供的第三方目标")
endif()
