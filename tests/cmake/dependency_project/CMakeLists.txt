cmake_minimum_required(VERSION 3.23)
project(caors_dependency_contract LANGUAGES C CXX)
set(CAORS_BUILD_TESTING ON CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(CAORS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CAORS_BUILD_EXAMPLE_TESTING OFF CACHE BOOL "" FORCE)
add_library(host_catch INTERFACE)
add_library(Catch2::Catch2 ALIAS host_catch)
target_include_directories(host_catch INTERFACE "${CAORS_SOURCE_DIR}/3rd/catch-v3.15.0")
if(NOT CAORS_DEPENDENCY_CASE STREQUAL "base_only")
  add_library(host_catch_main STATIC "${CAORS_SOURCE_DIR}/3rd/catch-v3.15.0/catch2/catch_amalgamated.cpp")
  add_library(Catch2::Catch2WithMain ALIAS host_catch_main)
  target_link_libraries(host_catch_main PUBLIC Catch2::Catch2)
endif()
if(CAORS_DEPENDENCY_CASE STREQUAL "complete")
  set(Catch2_VERSION 3.15.0)
elseif(CAORS_DEPENDENCY_CASE STREQUAL "incompatible")
  set(Catch2_VERSION 2.13.0)
endif()
add_subdirectory("${CAORS_SOURCE_DIR}" caors)
if(TARGET caors_catch OR TARGET caors_catch_main)
  message(FATAL_ERROR "不应重复创建已有 Catch2")
endif()
