file(GLOB CAORS_TEST_DIRECTORIES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*")
list(SORT CAORS_TEST_DIRECTORIES)

foreach(CAORS_TEST_DIRECTORY IN LISTS CAORS_TEST_DIRECTORIES)
  if(
    IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${CAORS_TEST_DIRECTORY}"
    AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${CAORS_TEST_DIRECTORY}/CMakeLists.txt"
  )
    add_subdirectory("${CAORS_TEST_DIRECTORY}")
  endif()
endforeach()
